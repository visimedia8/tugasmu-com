<!-- Interactive Client Script for Filtering & Search -->
<script>
    (function() {
      const searchInput = document.getElementById('toolSearchInput');
      const categoryBtns = document.querySelectorAll('.category-btn');
      const cards = document.querySelectorAll('.tool-card');
      const resultsCountBadge = document.getElementById('resultsCountBadge');
      const noResultsState = document.getElementById('noResultsState');
      const resetFiltersBtn = document.getElementById('resetFiltersBtn');
      const togglePopular = document.getElementById('togglePopular');
      const toggleMerdeka = document.getElementById('toggleMerdeka');

      let currentCategory = 'all';
      let filterPopular = false;
      let filterMerdeka = false;

      function applyFilters() {
        const query = (searchInput.value || '').trim().toLowerCase();
        let visibleCount = 0;

        cards.forEach(card => {
          const name = (card.getAttribute('data-name') || '').toLowerCase();
          const categories = (card.getAttribute('data-category') || '').toLowerCase();
          const isPop = card.getAttribute('data-popular') === 'true';
          const isMdk = card.getAttribute('data-merdeka') === 'true';

          const matchesQuery = !query || name.includes(query) || categories.includes(query);
          const matchesCategory = (currentCategory === 'all') || categories.includes(currentCategory);
          const matchesPopular = !filterPopular || isPop;
          const matchesMerdeka = !filterMerdeka || isMdk;

          if (matchesQuery && matchesCategory && matchesPopular && matchesMerdeka) {
            card.style.display = 'flex';
            visibleCount++;
          } else {
            card.style.display = 'none';
          }
        });

        resultsCountBadge.textContent = visibleCount + ' Tools Aktif';

        if (visibleCount === 0) {
          noResultsState.classList.remove('hidden');
          noResultsState.classList.add('flex');
        } else {
          noResultsState.classList.add('hidden');
          noResultsState.classList.remove('flex');
        }
      }

      // Search input binding
      searchInput.addEventListener('input', applyFilters);

      // Keyboard shortcut Ctrl+K
      window.addEventListener('keydown', (e) => {
        if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
          e.preventDefault();
          searchInput.focus();
          searchInput.select();
        }
      });

      // Category Pill Click
      categoryBtns.forEach(btn => {
        btn.addEventListener('click', () => {
          categoryBtns.forEach(b => {
            b.classList.remove('bg-primary-container', 'text-on-primary', 'shadow-sm');
            b.classList.add('bg-surface-container-low', 'text-on-surface-variant');
          });

          btn.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
          btn.classList.add('bg-primary-container', 'text-on-primary', 'shadow-sm');

          currentCategory = btn.getAttribute('data-filter') || 'all';
          applyFilters();
        });
      });

      // Quick Toggles
      togglePopular.addEventListener('click', () => {
        filterPopular = !filterPopular;
        if (filterPopular) {
          togglePopular.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
          togglePopular.classList.add('bg-tertiary-fixed', 'text-on-tertiary-fixed', 'font-bold');
        } else {
          togglePopular.classList.add('bg-surface-container-low', 'text-on-surface-variant');
          togglePopular.classList.remove('bg-tertiary-fixed', 'text-on-tertiary-fixed', 'font-bold');
        }
        applyFilters();
      });

      toggleMerdeka.addEventListener('click', () => {
        filterMerdeka = !filterMerdeka;
        if (filterMerdeka) {
          toggleMerdeka.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
          toggleMerdeka.classList.add('bg-secondary-fixed', 'text-on-secondary-fixed', 'font-bold');
        } else {
          toggleMerdeka.classList.add('bg-surface-container-low', 'text-on-surface-variant');
          toggleMerdeka.classList.remove('bg-secondary-fixed', 'text-on-secondary-fixed', 'font-bold');
        }
        applyFilters();
      });

      // Reset Button
      resetFiltersBtn.addEventListener('click', () => {
        searchInput.value = '';
        currentCategory = 'all';
        filterPopular = false;
        filterMerdeka = false;

        // Reset UI buttons
        categoryBtns.forEach(b => {
          b.classList.remove('bg-primary-container', 'text-on-primary', 'shadow-sm');
          b.classList.add('bg-surface-container-low', 'text-on-surface-variant');
        });
        const defaultBtn = document.querySelector('[data-filter="all"]');
        if (defaultBtn) {
          defaultBtn.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
          defaultBtn.classList.add('bg-primary-container', 'text-on-primary', 'shadow-sm');
        }

        togglePopular.classList.add('bg-surface-container-low', 'text-on-surface-variant');
        togglePopular.classList.remove('bg-tertiary-fixed', 'text-on-tertiary-fixed', 'font-bold');

        toggleMerdeka.classList.add('bg-surface-container-low', 'text-on-surface-variant');
        toggleMerdeka.classList.remove('bg-secondary-fixed', 'text-on-secondary-fixed', 'font-bold');

        applyFilters();
      });

    })();
  </script>
